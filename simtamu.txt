const ACCOUNTS = {

/* ---------------- TABS ---------------- */
function switchTab(tab){
  document.querySelectorAll('.tab-btn').forEach(b => b.classList.toggle('active', b.dataset.tab === tab));
  document.getElementById('tab-input').style.display = tab === 'input' ? 'block' : 'none';
  document.getElementById('tab-data').style.display = tab === 'data' ? 'block' : 'none';
  document.getElementById('tab-laporan').style.display = tab === 'laporan' ? 'block' : 'none';
  if(tab === 'data') renderDataTable();
  if(tab === 'laporan') renderLaporan();
}

/* ---------------- DATA (persisted, shared across petugas) ---------------- */
const STORAGE_KEY = 'buku-tamu-data';
let tamuList = [];

async function loadData(){
  try{
    const res = await window.storage.get(STORAGE_KEY, true);
    tamuList = res && res.value ? JSON.parse(res.value) : [];
  }catch(e){
    tamuList = [];
  }
}

async function saveData(){
  try{
    await window.storage.set(STORAGE_KEY, JSON.stringify(tamuList), true);
  }catch(e){
    showToast('Gagal menyimpan data. Coba lagi.');
  }
}

function showToast(msg){
  const t = document.getElementById('toast');
  t.textContent = msg;
  t.style.display = 'block';
  clearTimeout(window._toastTimer);
  window._toastTimer = setTimeout(() => t.style.display = 'none', 2400);
}

async function submitTamu(){
  const nama = document.getElementById('f-nama').value.trim();
  const instansi = document.getElementById('f-instansi').value.trim();
  const alamat = document.getElementById('f-alamat').value.trim();
  const hp = document.getElementById('f-hp').value.trim();
  const bertemu = document.getElementById('f-bertemu').value.trim();
  const keperluan = document.getElementById('f-keperluan').value.trim();

  if(!nama || !keperluan){
    showToast('Nama tamu dan keperluan wajib diisi.');
    return;
  }

  const now = new Date();
  const entry = {
    id: Date.now().toString(36) + Math.random().toString(36).slice(2,6),
    tanggalISO: now.toISOString(),
    nama, instansi, alamat, hp, bertemu, keperluan,
    petugas: currentUser.label
  };
  tamuList.push(entry);
  await saveData();
  showToast('Data tamu berhasil disimpan.');

  ['f-nama','f-instansi','f-alamat','f-hp','f-bertemu','f-keperluan'].forEach(id => document.getElementById(id).value = '');
}

function fmtTanggal(iso){
  const d = new Date(iso);
  return d.toLocaleDateString('id-ID', { day:'2-digit', month:'short', year:'numeric' }) + ' ' +
         d.toLocaleTimeString('id-ID', { hour:'2-digit', minute:'2-digit' });
}

function renderDataTable(){
  const q = (document.getElementById('search-box').value || '').toLowerCase();
  const tbody = document.getElementById('data-tbody');
  const sorted = [...tamuList].sort((a,b) => new Date(b.tanggalISO) - new Date(a.tanggalISO));
  const filtered = sorted.filter(t =>
    t.nama.toLowerCase().includes(q) || (t.instansi||'').toLowerCase().includes(q)
  );
  tbody.innerHTML = '';
  document.getElementById('data-empty').style.display = filtered.length ? 'none' : 'block';
  filtered.forEach((t, i) => {
    const tr = document.createElement('tr');
    tr.innerHTML = `
      <td class="mono">${i+1}</td>
      <td class="mono">${fmtTanggal(t.tanggalISO)}</td>
      <td>${escapeHtml(t.nama)}</td>
      <td>${escapeHtml(t.instansi||'—')}</td>
      <td class="mono">${escapeHtml(t.hp||'—')}</td>
      <td>${escapeHtml(t.keperluan)}</td>
      <td>${escapeHtml(t.bertemu||'—')}</td>
      <td>${escapeHtml(t.petugas)}</td>
      <td><button class="del-btn" onclick="hapusTamu('${t.id}')">Hapus</button></td>
    `;
    tbody.appendChild(tr);
  });
}

async function hapusTamu(id){
  if(!confirm('Hapus data tamu ini?')) return;
  tamuList = tamuList.filter(t => t.id !== id);
  await saveData();
  renderDataTable();
  showToast('Data tamu dihapus.');
}

function escapeHtml(str){
  const div = document.createElement('div');
  div.textContent = str ?? '';
  return div.innerHTML;
}

/* ---------------- LAPORAN BULANAN ---------------- */
const NAMA_BULAN = ['Januari','Februari','Maret','April','Mei','Juni','Juli','Agustus','September','Oktober','November','Desember'];

function initLaporanSelectors(){
  const bulanSel = document.getElementById('lap-bulan');
  const tahunSel = document.getElementById('lap-tahun');
  const now = new Date();
  bulanSel.innerHTML = NAMA_BULAN.map((b,i) => `<option value="${i}" ${i===now.getMonth()?'selected':''}>${b}</option>`).join('');
  const startYear = now.getFullYear() - 3;
  let opts = '';
  for(let y = now.getFullYear() + 1; y >= startYear; y--){
    opts += `<option value="${y}" ${y===now.getFullYear()?'selected':''}>${y}</option>`;
  }
  tahunSel.innerHTML = opts;
}

function getLaporanFiltered(){
  const bulan = parseInt(document.getElementById('lap-bulan').value, 10);
  const tahun = parseInt(document.getElementById('lap-tahun').value, 10);
  return tamuList
    .filter(t => {
      const d = new Date(t.tanggalISO);
      return d.getMonth() === bulan && d.getFullYear() === tahun;
    })
    .sort((a,b) => new Date(a.tanggalISO) - new Date(b.tanggalISO));
}

function renderLaporan(){
  const data = getLaporanFiltered();
  const tbody = document.getElementById('lap-tbody');
  tbody.innerHTML = '';
  document.getElementById('lap-empty').style.display = data.length ? 'none' : 'block';
  data.forEach((t, i) => {
    const tr = document.createElement('tr');
    tr.innerHTML = `
      <td class="mono">${i+1}</td>
      <td class="mono">${fmtTanggal(t.tanggalISO)}</td>
      <td>${escapeHtml(t.nama)}</td>
      <td>${escapeHtml(t.instansi||'—')}</td>
      <td>${escapeHtml(t.keperluan)}</td>
      <td>${escapeHtml(t.bertemu||'—')}</td>
    `;
    tbody.appendChild(tr);
  });
  document.getElementById('stat-total').textContent = data.length;
  const uniqInstansi = new Set(data.map(t => (t.instansi||'').trim().toLowerCase()).filter(Boolean));
  document.getElementById('stat-instansi').textContent = uniqInstansi.size;
}

function cetakLaporan(){
  const data = getLaporanFiltered();
  const bulanIdx = parseInt(document.getElementById('lap-bulan').value, 10);
  const tahun = document.getElementById('lap-tahun').value;
  document.getElementById('print-periode').textContent = `Periode: ${NAMA_BULAN[bulanIdx]} ${tahun}`;
  document.getElementById('print-nama-petugas').textContent = currentUser ? currentUser.label : '—';
  document.getElementById('print-tanggal-cetak').textContent = new Date().toLocaleDateString('id-ID', { day:'2-digit', month:'long', year:'numeric' });

  const tbody = document.getElementById('print-tbody');
  tbody.innerHTML = '';
  if(data.length === 0){
    tbody.innerHTML = `<tr><td colspan="8" style="text-align:center; padding:16px; color:#5B6472;">Tidak ada data kunjungan pada periode ini.</td></tr>`;
  } else {
    data.forEach((t, i) => {
      const tr = document.createElement('tr');
      tr.innerHTML = `
        <td>${i+1}</td>
        <td>${fmtTanggal(t.tanggalISO)}</td>
        <td>${escapeHtml(t.nama)}</td>
        <td>${escapeHtml(t.instansi||'—')}</td>
        <td>${escapeHtml(t.alamat||'—')}</td>
        <td>${escapeHtml(t.hp||'—')}</td>
        <td>${escapeHtml(t.keperluan)}</td>
        <td>${escapeHtml(t.bertemu||'—')}</td>
      `;
      tbody.appendChild(tr);
    });
  }
  window.print();
}
